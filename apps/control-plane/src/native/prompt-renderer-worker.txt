// Runs only in the private WorkerLoader isolate; the application never starts Go.
import './wasm_exec.js';
import rendererModule from './prompt-renderer.wasm';

let ready;
let instance;
let failure;
function initialize() {
  return ready ??= (async () => {
    const go = new globalThis.Go();
    instance = await WebAssembly.instantiate(rendererModule, go.importObject);
    void go.run(instance).catch(error => { failure = error; });
    if (!globalThis.__pentcloudRender) throw new Error('Go prompt renderer did not initialize');
  })();
}
export default {
  async fetch(request) {
    await initialize();
    if (failure) throw failure;
    if (new URL(request.url).pathname === '/memory') {
      return Response.json({ linearMemoryBytes: instance.exports.mem.buffer.byteLength });
    }
    const input = await request.text();
    return new Response(globalThis.__pentcloudRender(input), {
      headers: { 'Content-Type': 'application/json' },
    });
  },
};
